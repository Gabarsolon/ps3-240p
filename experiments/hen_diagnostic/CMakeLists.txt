cmake_minimum_required(VERSION 3.20)
project(crt-hen-diagnostic C ASM)
option(HD_PUBLIC_ONLY "Exclude all custom syscall-8/HEN probes" OFF)
include(ps3-self)
add_executable(crt-hen-diagnostic main.c diagnostic.c native.c video_imports.S)
target_compile_features(crt-hen-diagnostic PRIVATE c_std_11)
if(HD_PUBLIC_ONLY)
    target_compile_definitions(crt-hen-diagnostic PRIVATE HD_PUBLIC_ONLY=1)
endif()
target_compile_options(crt-hen-diagnostic PRIVATE -O2 -Wall -Wextra -Werror)
target_link_options(crt-hen-diagnostic PRIVATE -Wl,--gc-sections)
target_link_libraries(crt-hen-diagnostic PRIVATE lv2)
ps3_add_self(crt-hen-diagnostic OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
